# --- Spot-assignment strategy demo (heap-based nearest lookup) ---
# All 4 MEDIUM spots free: F1-M1(exit2,elev4) F1-M2(exit4,elev2) F2-M1(exit1,elev4) F2-M2(exit4,elev1)

# ELEVATOR preference should pick the free MEDIUM spot with smallest elevator distance -> F2-M2 (elev=1)
PARK CAR KA01C1111 0 ELEVATOR

# EXIT preference should pick the smallest remaining exit distance -> F2-M1 (exit=1)
PARK CAR KA01C2222 0 EXIT

# ANY (no preference) reuses the exit-ordered heap -> next smallest exit distance -> F1-M1 (exit=2)
PARK CAR KA01C3333 0 ANY

# No preference token at all defaults to ANY -> only F1-M2 left among MEDIUM -> F1-M2
PARK CAR KA01C4444 0

# MEDIUM now exhausted; falls through to LARGE. ANY picks smallest exit distance among free LARGE -> F2-L1 (exit=1)
PARK CAR KA01C5555 0 ANY

STATUS

# Release one and confirm it becomes selectable again via either heap
UNPARK KA01C1111 30
PARK CAR KA01C6666 40 ELEVATOR
